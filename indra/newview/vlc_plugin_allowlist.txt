# libVLC plugin allowlist for the Windows package (media plugin hardening:
# ship only the stream access modules we use).
#
# viewer_manifest.py copies EXACTLY these files from <packages>/bin/release/plugins
# into <app>/plugins and fails the packaging step if any listed file is missing.
# Anything not listed here is not shipped. One path per line, relative to plugins/,
# forward slashes. Lines starting with '#' are comments.
#
# Adding a module: add the line AND a justification in its group. Prefer
# leaving things out; libVLC skips optional modules that are absent.

# --- Stream access: network only -------------------------------------------
# http + https are the only schemes the viewer hands to the media plugin
# (parcel audio/video streams, MOAP media). No local-file, share, discovery or
# capture access modules are shipped.
access/libhttp_plugin.dll
access/libhttps_plugin.dll
# TLS client for https.
misc/libgnutls_plugin.dll
# In-memory credential store used by the http access module (no disk I/O).
keystore/libmemory_keystore_plugin.dll

# --- Stream filters ---------------------------------------------------------
# Read-ahead / block caching for network streams, and ID3v2 tag skipping so
# mp3 streams with leading tags probe correctly.
stream_filter/libprefetch_plugin.dll
stream_filter/libcache_read_plugin.dll
stream_filter/libcache_block_plugin.dll
stream_filter/libskiptags_plugin.dll

# --- Demuxers ---------------------------------------------------------------
# Common stream and video containers. adaptive = HLS/DASH (segments are
# fetched through the same http/https access modules). es = raw elementary
# streams (mp3/aac/ac3 radio). No playlist demuxer (m3u/pls/xspf): the media
# plugin plays the single URL it is given and never follows sub-items, and a
# playlist demuxer would only add a way to chain to further URLs.

demux/libes_plugin.dll
demux/libmp4_plugin.dll

# --- Audio output path ------------------------------------------------------
# Parcel audio capture (ParcelAudioCapture, #892) hands the decoded PCM to the
# viewer through libvlc_audio_set_callbacks, which forces the player's audio
# output to "amem,none": without amem there is NO audio output at all (the
# stream demuxes and shows its title, but nothing reaches the viewer and
# nothing plays; #960). amem writes to memory only, it opens no device.
audio_output/libamem_plugin.dll
# Without capture (capture off, parcel media, or the viewer's fallback when
# capture delivers nothing) audio plays to the system device: mmdevice is the
# default Windows output and loads wasapi as its stream backend;
# directsound/waveout are its fallbacks; adummy = no-device fallback.
# Mixers apply volume; the filters below are format/rate/channel conversion.
audio_output/libmmdevice_plugin.dll
audio_output/libwasapi_plugin.dll
audio_output/libdirectsound_plugin.dll
audio_output/libwaveout_plugin.dll
audio_output/libadummy_plugin.dll
